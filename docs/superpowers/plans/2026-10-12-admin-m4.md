```tsx
'use client'

import { useQuery } from '@tanstack/react-query'
import { Avatar } from '@xsl/lomva-ui/avatar'
import { cn } from '@xsl/lomva-ui/cn'
import { IconButton } from '@xsl/lomva-ui/icon-button'
import { useTranslation } from 'react-i18next'
import { userProfileQueryOptions } from '@/features/account-profile/client'
import Link from '@/next/link'
import { usePathname, useRouter } from '@/next/navigation'
import { useLogout } from '@/service/use-common'

function navLinkClassName(active: boolean) {
  return cn(
    'flex h-9 items-center rounded-lg px-3 text-sm text-text-secondary hover:bg-state-base-hover',
    active && 'bg-state-base-hover font-medium text-text-accent',
  )
}

export function AdminSidebar() {
  const { t } = useTranslation()
  const pathname = usePathname()
  const router = useRouter()
  const { data } = useQuery(userProfileQueryOptions())
  const { mutateAsync: logout, isPending: isLoggingOut } = useLogout()

  const handleLogout = async () => {
    await logout()
    router.replace('/signin')
  }

  const workspacesActive = pathname.startsWith('/admin/workspaces')
  const auditActive = pathname.startsWith('/admin/audit-logs')

  return (
    <aside className="flex h-full w-56 shrink-0 flex-col border-r border-divider-subtle bg-background-default-subtle">
      <div className="px-4 pb-2 pt-4 text-sm font-semibold text-text-primary">
        {t(($) => $.nav.title, { ns: 'admin' })}
      </div>
      <nav className="flex flex-col gap-0.5 px-2">
        <Link
          href="/admin/workspaces"
          aria-current={workspacesActive ? 'page' : undefined}
          className={navLinkClassName(workspacesActive)}
        >
          {t(($) => $.nav.workspaces, { ns: 'admin' })}
        </Link>
        <Link
          href="/admin/audit-logs"
          aria-current={auditActive ? 'page' : undefined}
          className={navLinkClassName(auditActive)}
        >
          {t(($) => $.nav.auditLogs, { ns: 'admin' })}
        </Link>
      </nav>
      <div className="mt-auto flex items-center gap-2 border-t border-divider-subtle p-3">
        <Avatar
          avatar={data?.profile.avatar_url ?? null}
          name={data?.profile.name ?? ''}
          size="sm"
          className="shrink-0"
        />
        <span className="min-w-0 grow truncate text-sm text-text-primary">{data?.profile.name}</span>
        <IconButton
          aria-label={t(($) => $['userProfile.logout'], { ns: 'common' })}
          disabled={isLoggingOut}
          onClick={() => void handleLogout()}
        >
          <span aria-hidden className="i-ri-logout-box-r-line size-4" />
        </IconButton>
      </div>
    </aside>
  )
}
```

```tsx
'use client'

import type { ReactNode } from 'react'
import { useQuery } from '@tanstack/react-query'
import { useTranslation } from 'react-i18next'
import Loading from '@/app/components/base/loading'
import { userProfileQueryOptions } from '@/features/account-profile/client'
import Link from '@/next/link'
import { AdminShell } from './admin-shell'

export function AdminGuard({ children }: { children: ReactNode }) {
  const { t } = useTranslation()
  const { data, isPending } = useQuery(userProfileQueryOptions())

  if (isPending) {
    return (
      <div className="flex h-screen w-full items-center justify-center">
        <Loading />
      </div>
    )
  }

  if (!data?.profile.is_system_admin) {
    return (
      <div className="flex h-screen w-full flex-col items-center justify-center gap-3 px-6 text-center">
        <h1 className="text-xl font-semibold text-text-primary">
          {t(($) => $.guard.forbiddenTitle, { ns: 'admin' })}
        </h1>
        <p className="text-sm text-text-tertiary">
          {t(($) => $.guard.forbiddenDesc, { ns: 'admin' })}
        </p>
        <Link href="/" className="text-sm text-text-accent">
          {t(($) => $.guard.backHome, { ns: 'admin' })}
        </Link>
      </div>
    )
  }

  return <AdminShell>{children}</AdminShell>
}
```

```tsx
'use client'

import { useMutation, useQueryClient } from '@tanstack/react-query'
import { Button } from '@xsl/lomva-ui/button'
import {
  Dialog,
  DialogBackdrop,
  DialogPopup,
  DialogPortal,
  DialogTitle,
} from '@xsl/lomva-ui/dialog'
import { Input } from '@xsl/lomva-ui/input'
import { toast } from '@xsl/lomva-ui/toast'
import { useState } from 'react'
import { useTranslation } from 'react-i18next'
import { consoleQuery } from '@/service/client'

type CreateWorkspaceDialogProps = {
  open: boolean
  onOpenChange: (open: boolean) => void
}

export function CreateWorkspaceDialog({ open, onOpenChange }: CreateWorkspaceDialogProps) {
  const { t } = useTranslation()
  const queryClient = useQueryClient()
  const [name, setName] = useState('')
  const [ownerEmail, setOwnerEmail] = useState('')
  // PENDING 分支结果态:非 null 即展示邀请链接页
  const [pendingResult, setPendingResult] = useState<{ inviteUrl: string | null } | null>(null)

  const reset = () => {
    setName('')
    setOwnerEmail('')
    setPendingResult(null)
  }

  const close = () => {
    onOpenChange(false)
    reset()
  }

  const createMutation = useMutation({
    ...consoleQuery.admin.workspaces.post.mutationOptions(),
    onSuccess: async (res) => {
      await queryClient.invalidateQueries({
        queryKey: consoleQuery.admin.workspaces.get.queryKey(),
      })
      if (res.owner_pending) {
        setPendingResult({ inviteUrl: res.invite_url ?? null })
      }
      else {
        toast.success(t(($) => $.workspaces.createSuccess, { ns: 'admin' }))
        close()
      }
    },
    onError: () => {
      toast.error(t(($) => $.workspaces.createFailed, { ns: 'admin' }))
    },
  })

  const onSubmit = (event: React.FormEvent<HTMLFormElement>) => {
    event.preventDefault()
    if (createMutation.isPending) return
    createMutation.mutate({ body: { name: name.trim(), owner_email: ownerEmail.trim() } })
  }

  const absoluteInviteUrl = pendingResult?.inviteUrl
    ? `${window.location.origin}${pendingResult.inviteUrl}`
    : null

  const copyInvite = async () => {
    if (!absoluteInviteUrl) return
    await navigator.clipboard.writeText(absoluteInviteUrl)
    toast.success(t(($) => $.workspaces.copied, { ns: 'admin' }))
  }

  return (
    <Dialog open={open} onOpenChange={(next) => (next ? onOpenChange(true) : close())}>
      <DialogPortal>
        <DialogBackdrop />
        <DialogPopup className="fixed left-1/2 top-1/2 w-120 max-w-[calc(100vw-2rem)] -translate-x-1/2 -translate-y-1/2 p-6">
          {pendingResult === null && (
            <>
              <DialogTitle className="text-base font-semibold text-text-primary">
                {t(($) => $.workspaces.createTitle, { ns: 'admin' })}
              </DialogTitle>
              <form className="mt-4 flex flex-col gap-4" onSubmit={onSubmit}>
                <label className="flex flex-col gap-1.5 text-sm text-text-secondary">
                  {t(($) => $.workspaces.createNameLabel, { ns: 'admin' })}
                  <Input
                    required
                    value={name}
                    onChange={(event) => setName(event.target.value)}
                    placeholder={t(($) => $.workspaces.createNamePlaceholder, { ns: 'admin' })}
                  />
                </label>
                <label className="flex flex-col gap-1.5 text-sm text-text-secondary">
                  {t(($) => $.workspaces.createOwnerLabel, { ns: 'admin' })}
                  <Input
                    required
                    type="email"
                    value={ownerEmail}
                    onChange={(event) => setOwnerEmail(event.target.value)}
                    placeholder={t(($) => $.workspaces.createOwnerPlaceholder, { ns: 'admin' })}
                  />
                </label>
                <p className="text-xs text-text-tertiary">
                  {t(($) => $.workspaces.createHint, { ns: 'admin' })}
                </p>
                <div className="flex justify-end gap-2">
                  <Button type="button" variant="secondary" onClick={close}>
                    {t(($) => $.workspaces.createCancel, { ns: 'admin' })}
                  </Button>
                  <Button type="submit" variant="primary" disabled={createMutation.isPending}>
                    {t(($) => $.workspaces.createSubmit, { ns: 'admin' })}
                  </Button>
                </div>
              </form>
            </>
          )}
          {pendingResult !== null && (
            <>
              <DialogTitle className="text-base font-semibold text-text-primary">
                {t(($) => $.workspaces.pendingTitle, { ns: 'admin' })}
              </DialogTitle>
              <p className="mt-4 text-sm text-text-secondary">
                {t(($) => $.workspaces.pendingMailSent, { ns: 'admin' })}
```

```tsx
export default function AdminAuditLogsPage() {

              <SelectItemText>{t(($) => $.audit.filterWorkspaceAll, { ns: 'admin' })}</SelectItemText>
            </SelectItem>
```

```tsx
'use client'

import { useQuery } from '@tanstack/react-query'
import { useAtomValue } from 'jotai'
import { useEffect } from 'react'
import { currentWorkspaceAtom, currentWorkspaceLoadingAtom } from '@/context/workspace-state'
import { userProfileQueryOptions } from '@/features/account-profile/client'
import { useRouter } from '@/next/navigation'

/**
 * 无空间系统管理员兜底:登录落地 / 与误入普通控制台页两场景统一 → /admin。
 * 仅在 profile 与 workspace 两个 query 均落地后判定,避免加载期误跳。
 * 反向无循环:/admin 守卫对非管理员只渲染 403 视图、不重定向。
 */
export function AdminLandingRedirect() {
  const router = useRouter()
  const { data: profileData, isPending: profilePending } = useQuery(userProfileQueryOptions())
  const currentWorkspace = useAtomValue(currentWorkspaceAtom)
  const workspacePending = useAtomValue(currentWorkspaceLoadingAtom)

  useEffect(() => {
    if (profilePending || workspacePending) return
    if (profileData?.profile.is_system_admin && !currentWorkspace.id)
      router.replace('/admin')
  }, [profilePending, workspacePending, profileData, currentWorkspace.id, router])

  return null
}
```
