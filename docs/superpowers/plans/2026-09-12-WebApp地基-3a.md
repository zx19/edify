- [ ] **Step 3: 实现 client**

Create `web-new/service/webapp-client.ts`：

```ts
import type { ContractRouterClient } from '@orpc/contract'
import type { JsonifiedClient } from '@orpc/openapi-client'
import { createORPCClient, onError } from '@orpc/client'
import { OpenAPILink } from '@orpc/openapi-client/fetch'
import { contract as webappContract } from '@dify/contracts/api/web/orpc.gen'
import { PASSPORT_HEADER_NAME, PUBLIC_API_PREFIX } from '@/config'
import { request } from './base'
import { getBaseURL } from './client'
import { resolveWebAppAddress } from './webapp-address'
import { getWebAppPassport } from './webapp-auth'

type WebAppClientContext = { silent?: boolean }
type WebAppClientLink = ClientLink<WebAppClientContext>

function createWebAppOpenAPILink(): WebAppClientLink {
  return new OpenAPILink<WebAppClientContext>(webappContract, {
    url: getBaseURL(PUBLIC_API_PREFIX),
    headers: () => {
      const passport = getWebAppPassport(resolveWebAppAddress())
      return passport ? { [PASSPORT_HEADER_NAME]: passport } : {}
    },
    fetch: (input, _init, options) => {
      // 剥回相对路径交还 base request：formatURL(isPublicAPI) 内部重做
      // PUBLIC_API_PREFIX 前缀 + getWebAppPublicApiPath 地址解析（env 形态等）
      const linkURL = new URL(getBaseURL(PUBLIC_API_PREFIX))
      const path = new URL(input.url).pathname.replace(linkURL.pathname.replace(/\/$/, ''), '')
      return request(path, { headers: input.headers, method: input.method }, {
        isPublicAPI: true,
        fetchCompat: true,
        silent: options.context?.silent,
      })
    },
    interceptors: [
      onError((error) => {
        console.error(error)
      }),
    ],
  })
}

export const webappClient: JsonifiedClient<
  ContractRouterClient<typeof webappContract, WebAppClientContext>
> = createORPCClient(createWebAppOpenAPILink())
```

注意点（执行时对 `client.ts` 的 console 范式校准，tsc 裁判）：
- `ClientLink` 类型 import 自 `@orpc/client`（与 client.ts 一致）。
- body/query 透传：本计划只消费 GET 无 body；`fetch` 适配器先满足 GET 形态，POST/SSE 端点不在 3a（第 4 步扩展时补 body 序列化透传并补测试）。
- 若 OpenAPILink 要求 `fetch` 返回 `Response`：`request` 带 `fetchCompat: true` 即此语义（console link 同用法实证）。
